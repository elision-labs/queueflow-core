Check it:

  kubectl -n {{ .Release.Namespace }} rollout status deploy/{{ include "queueflow.apiDeploymentName" . }}
  kubectl -n {{ .Release.Namespace }} port-forward svc/{{ include "queueflow.fullname" . }} 8000:{{ .Values.service.port }} &
  curl -s localhost:8000/health      # {"status":"ok",...} once Postgres is reachable
  curl -s localhost:8000/ready
  open http://localhost:8000/docs    # Swagger UI

Then enqueue a job with one of your tenant API keys:

  curl -s -X POST localhost:8000/api/v1/jobs \
    -H "Authorization: Bearer <token>" -H 'Content-Type: application/json' \
    -d '{"task_name":"echo","payload":{"hello":"world"}}'
